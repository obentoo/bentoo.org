# 源 — stage 1 · download & verify
wget -c https://obentoo.org/stage3-latest.tar.xz
gpg --verify stage3-latest.tar.xz.sig

# 入 — stage 2/3 · chroot & profile
tar xpf stage3-latest.tar.xz -C /mnt/gentoo
chroot /mnt/gentoo /bin/bash
eselect profile set bentoo/desktop/plasma

# 世界 — stage 4 · rebuild world
emerge -avuDN @world
